You are given an array of integers nums, there is a sliding window of size k which is moving from the very left of the array 
to the very right. You can only see the k numbers in the window. Each time the sliding window moves right by one position.
Return the max sliding window.

Input: nums = [1,3,-1,-3,5,3,6,7], k = 3
Output: [3,3,5,5,6,7]
Explanation: 
Window position                Max
---------------               -----
[1  3  -1] -3  5  3  6  7       3
 1 [3  -1  -3] 5  3  6  7       3
 1  3 [-1  -3  5] 3  6  7       5
 1  3  -1 [-3  5  3] 6  7       5
 1  3  -1  -3 [5  3  6] 7       6
 1  3  -1  -3  5 [3  6  7]      7

 class Solution{
    public int[] maxSlidingWindow(int[] nums,int k){
        int n=nums.length;
        int[] result=new int[n-k+1];
        Deque<Integer> dq=new ArrayDeque<>();

        for(int i=0;i<k;i++){
            while(!dq.isEmpty() && nums[dq.peekLast()]<nums[i]){
                dq.pollLast();
            }
            dq.add(i);
        }
        result[0]=nums[dq.peekFirst()];

        for(int i=k;i<n;i++){
            if(dq.peekFirst()<=i-k){
                dq.poll();
            }
            while(!dq.isEmpty() && nums[dq.peekLast()]<=nums[i]){
                dq.pollLast();
            }
            dq.add(i);

            result[i-k+1]=nums[dq.peek()];
        }
        return result;
    }
 }
