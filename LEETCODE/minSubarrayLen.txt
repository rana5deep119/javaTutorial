class Solution{
    public int minSubArrayLen(int[] nums, int target){
        int start=0;
        int minLen=0;
        int sum=0;

        for(int end=0;end<nums.length;end++){
            sum+=nums[end];
            while(sum>=target){
                minLen=Math.min(minLen,end-start+1);
                sum-=nums[start];
                start++;
            }
        }
          return minLen;
    }
}
