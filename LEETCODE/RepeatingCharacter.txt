424. Longest Repeating Character Replacement

You are given a string s and an integer k. You can choose any character of the string and change it to any other uppercase 
English character. You can perform this operation at most k times. Return the length of the longest substring containing the 
same letter you can get after performing the above operations.

Input: s = "ABAB", k = 2
Output: 4
Explanation: Replace the two 'A's with two 'B's or vice versa.
Example 2:

Input: s = "AABABBA", k = 1
Output: 4
Explanation: Replace the one 'A' in the middle with 'B' and form "AABBBBA".
The substring "BBBB" has the longest repeating letters, which is 4.
 

class Solution{
    public int replacement(String s, int k){
        int maxlen=0;
        HashMap<Character,Integer> mp=new HashMap<>();
        int maxFreq=0;

        int l=0;
        for(int right=0;r<s.length();r++){
            char ch=s.charAt(r);
            mp.put(ch,mp.getOrDefault(ch,0)+1);
            maxFreq=Math.max(maxFreq,mp.get(ch));
            while((r-l+1)-maxFreq > k){
                char leftChar=s.charAt(l);
                mp.put(leftChar,mp.get(leftChar)-1);
                l++;
            }
            maxlen=Math.mac(maxlen,r-l+1);
        }
        return maxlen;
    }
}

class Solution{
    public int replacement(String s, int k){
        int maxlen=0;
        HashMap<Character,Integer> mp=new HashMap<>();
        int maxFreq=0;

        int l=0;
        for(int r=0;r<s.length;r++){
            char ch=s.charAt(r);
            mp.put(ch,mp.getOrDefault(ch,0)+1);   // we are putting the freq of each char while tranversing , in a map
            maxFreq=Math.max(maxFreq,mp.get(ch)); // find the maximum frequency from the map

            // we will check if windowSize-maxfreq greater than k , then we will reduce the window from left 
            while({r-l+1}-maxFreq>k){
                // reduce the window from left and reduce the left elemnt freq from map
                char leftChar=s.charAt(l);
                mp.put(leftChar,mp.get(leftChar)-1);
                l++;   // move the left pointer ahead
            }
            
            // now find the maximum length
            maxlen=Math.max(maxlen,r-l+1);
        }
    }
}
